% ------------------------------------------------------------
% Wichtig:
% Dieser Schalter verändert NICHT \setmainfont, \setsansfont oder \setmonofont.
% Er betrifft ausschließlich die FDI-Font-Makros:
% \KarolingischeMinuskel, \FDIOverlayFont, \ArialBOLDFont und \ArialFont.
%
% \FDISetLanguageFonts{sprache}{zuweisungen}
% hinterlegt die Font-Zuweisungen einer Sprache. Aufgerufen wird das
% in der Datei fonts_CODE.tex im jeweiligen Sprachordner. Sprachen ohne
% eigene Datei benutzen automatisch die Standardschriften.

\makeatletter

\newcommand{\FDISetLanguageFonts}[2]{%
  \expandafter\def\csname FDIfonts@#1\endcsname{#2}%
}

\newcommand{\FDISetupLanguageFonts}{%
  % Immer zuerst die Standardschriften setzen ...
  \let\KarolingischeMinuskel\FDIKarolingischeMinuskelDefault
  \let\FDIOverlayFont\FDIOverlayFontDefault
  \let\ArialBOLDFont\FDIArialBOLDFontDefault
  \let\ArialFont\FDIArialFontDefault
  % ... und danach die Sprachschriften, falls die Sprache eigene mitbringt.
  \@ifundefined{FDIfonts@\FDILanguage}{%
    \typeout{FDI: Language \FDILanguage\space uses the default FDI font macros.}%
  }{%
    \typeout{FDI: Language \FDILanguage\space switches FDI font macros only.}%
    \csname FDIfonts@\FDILanguage\endcsname
  }%
}

\makeatother

% ------------------------------------------------------------
% Whole comic in one language
% Usage: \comic{deu}        alle Comicseiten auf Deutsch
%        \comic[3]{eng}     nur die ersten 3 Seiten, also Seite 1 bis 3
%        \comic[3-5]{eng}   Seite 3 bis 5
%        \comic[7-7]{eng}   nur Seite 7
% ------------------------------------------------------------
% Gibt alle Comicseiten der Sprache #2 direkt hintereinander aus.
% Sprache, Fonts und Texte gelten nur innerhalb von \comic. Nach
% \comic gelten wieder die Einstellungen des Dokuments, also
% \FDILanguage sowie \title/\subtitle/\shorttitle aus der in der
% Praeambel geladenen Sprachdatei.
% Dadurch koennen auch mehrere Sprachen nacheinander ausgegeben
% werden, z.B.:
%   \comic{deu}
%   \comic{eng}

% Anzahl der Comicseiten pro Sprache
\newcommand{\FDIComicPageCount}{12}

\makeatletter

\newcounter{FDIcomicpage}

% Optionales Argument von \comic auswerten.
% Aufruf immer mit angehaengtem --\FDI@stop, dadurch gilt:
%   "5"    ->  #1=5  #2=leer  ->  Seite 1 bis 5
%   "3-5"  ->  #1=3  #2=5     ->  Seite 3 bis 5
\def\FDI@stop{}

\def\FDI@parserange#1-#2-#3\FDI@stop{%
  \def\FDI@tmp{#2}%
  \ifx\FDI@tmp\@empty
    \def\FDI@firstpage{1}%
    \edef\FDI@lastpage{#1}%
  \else
    \edef\FDI@firstpage{#1}%
    \edef\FDI@lastpage{#2}%
  \fi
}

\newcommand{\comic}[2][\FDIComicPageCount]{%
  \begingroup
    % \edef, damit auch \comic{\FDILanguage} funktioniert
    \edef\FDILanguage{#2}%
    \FDISetupLanguageFonts
    % Die Sprachdatei setzt am Anfang \title, \subtitle und \shorttitle.
    % Im Fliesstext duerfen die nichts tun: \maketitle hat \title global
    % auf \relax gesetzt, der Titel wuerde sonst als Text auf einer
    % eigenen Seite landen. Die Metadaten des Artikels stammen weiterhin
    % aus der Sprachdatei, die in der Praeambel geladen wird.
    \let\title\@gobble
    \let\subtitle\@gobble
    \let\shorttitle\@gobble
    \FDIInputLanguageFile
    % Seitenbereich bestimmen
    \edef\FDI@tmp{#1}%
    \expandafter\FDI@parserange\FDI@tmp--\FDI@stop
    % Beide Zaehler auf die Seite vor der ersten gewuenschten Seite setzen.
    % fdipage waehlt die Seite im Hintergrund-PDF und laeuft deshalb mit,
    % damit \comic[3-5] auch die Zeichnungen der Seiten 3 bis 5 zeigt.
    \setcounter{FDIcomicpage}{\FDI@firstpage}%
    \addtocounter{FDIcomicpage}{-1}%
    \setcounter{fdipage}{\value{FDIcomicpage}}%
    \@whilenum\value{FDIcomicpage}<\FDI@lastpage\do{%
      \stepcounter{FDIcomicpage}%
      \expandafter\FDIInputPage\expandafter{\number\value{FDIcomicpage}}%
      % Ueberschrift und Sprunganker liegen auf der ersten Comicseite,
      % nur wenn \FDIComicSection sie angefordert hat.
      \ifnum\value{FDIcomicpage}=\FDI@firstpage\relax
        \@ifundefined{FDI@comicheadingwanted}{}{\FDI@comicheading}%
      \fi
    }%
    % letzte Comicseite ausgeben, solange die Sprache noch gesetzt ist
    \clearpage
  \endgroup
}

\makeatother

% ------------------------------------------------------------
% Comic with heading and jump target
% Usage: \FDIComicSection{deu}
%        \FDIComicSection[3-5]{deu}
% ------------------------------------------------------------
% Gibt den kompletten Comic einer Sprache aus. Ueberschrift,
% Sprungmarke comic:sprache und PDF-Lesezeichen liegen auf der ersten
% Comicseite, damit keine fast leere Ueberschriftsseite entsteht.
% Gedacht fuer den Anhang, angesprungen wird die Marke von \FDIJumpTo.

\newcommand{\FDIComicSectionTitle}[1]{The comic in \FDILanguageName{#1}}

% Position der Ueberschrift auf der Comicseite, gemessen von oben links.
% Sie liegt im freien Band zwischen der Kopfzeile und der ersten
% Panel-Nummer, die die Seitendateien bei 2cm setzen.
\newcommand{\FDIComicHeadingXOffset}{2cm}
\newcommand{\FDIComicHeadingYOffset}{1.5cm}
\newcommand{\FDIComicHeadingFont}{\normalfont\small\bfseries}

\makeatletter

% Wird von \comic auf der ersten Comicseite ausgefuehrt.
\newcommand{\FDI@comicheading}{%
  % Abschnittsnummer (im Anhang A, B, C ...) und Sprungziel
  \refstepcounter{section}%
  \phantomsection
  \label{comic:\FDILanguage}%
  \addcontentsline{toc}{section}{%
    \protect\numberline{\thesection}\FDIComicSectionTitle{\FDILanguage}%
  }%
  % sichtbare Ueberschrift im oberen Seitenrand, ueber der Zeichnung
  \begin{tikzpicture}[remember picture, overlay]
    \node[
      anchor=north west,
      inner sep=0pt,
      outer sep=0pt
    ] at ([xshift=\FDIComicHeadingXOffset,yshift=-\FDIComicHeadingYOffset]current page.north west)
      {\FDIComicHeadingFont\thesection\quad\FDIComicSectionTitle{\FDILanguage}};
  \end{tikzpicture}%
}

\newcommand{\FDIComicSection}[2][\FDIComicPageCount]{%
  \clearpage
  \pagestyle{inggrid}%
  \nolinenumbers
  \begingroup
    % Signal an \comic: Ueberschrift und Anker auf die erste Comicseite
    \def\FDI@comicheadingwanted{1}%
    \comic[#1]{#2}%
  \endgroup
  % danach wieder normaler Satzzustand, damit Danksagung und
  % Literaturverzeichnis am Dokumentende aussehen wie gewohnt
  \clearpage
  \pagestyle{inggrid}%
  \@ifundefined{inggridlinenumbers}{}{\inggridlinenumbers}%
}

\makeatother

% ------------------------------------------------------------
% Jump to navigation
% Usage: \FDIJumpTo{article,eng,deu,lat,qya}
% ------------------------------------------------------------
% Zeigt eine Zeile mit Sprunglinks. Der Eintrag article springt zur
% Einleitung, jeder andere Eintrag zur Sprungmarke comic:sprache,
% also zu \FDIComicSection bzw. zum \label des Comics im Fliesstext.

\newcommand{\FDIJumpToTitle}{Jump to}
\newcommand{\FDIJumpToArticleName}{Article}
\newcommand{\FDIJumpToComicName}[1]{\FDILanguageName{#1} comic}
\newcommand{\FDIJumpToArticleLabel}{sec:introduction}
\newcommand{\FDIJumpToSeparator}{\hspace{0.6em}\textbar\hspace{0.6em}}

\makeatletter

\newif\ifFDI@firstjump

\newcommand{\FDI@jumpitem}[1]{%
  \def\FDI@jumptmp{#1}%
  \def\FDI@jumparticle{article}%
  \ifx\FDI@jumptmp\FDI@jumparticle
    \hyperref[\FDIJumpToArticleLabel]{\FDIJumpToArticleName}%
  \else
    \hyperref[comic:#1]{\FDIJumpToComicName{#1}}%
  \fi
}

\newcommand{\FDIJumpTo}[1]{%
  \begingroup
    \edef\FDI@jumplist{#1}%
    \FDI@firstjumptrue
    \noindent\textbf{\FDIJumpToTitle:}%
    \@for\FDI@jumpkey:=\FDI@jumplist\do{%
      \ifFDI@firstjump
        \FDI@firstjumpfalse
        \hspace{0.6em}%
      \else
        \FDIJumpToSeparator
      \fi
      \expandafter\FDI@jumpitem\expandafter{\FDI@jumpkey}%
    }%
    \par
  \endgroup
}

% \FDIJumpToAll
% Sprungleiste fuer alle Sprachen aus \FDILanguageList:
% zuerst der Artikel, dann der Comic im Fliesstext (\FDILanguage),
% danach die uebrigen Sprachen in der Reihenfolge der Liste.
\newcommand{\FDIJumpToAll}{%
  \edef\FDI@curlang{\FDILanguage}%
  \edef\FDI@jumpall{article,\FDILanguage}%
  \@for\FDI@lang:=\FDILanguageList\do{%
    \edef\FDI@langtmp{\FDI@lang}%
    \ifx\FDI@langtmp\FDI@curlang\else
      \edef\FDI@jumpall{\FDI@jumpall,\FDI@lang}%
    \fi
  }%
  \expandafter\FDIJumpTo\expandafter{\FDI@jumpall}%
}

% \FDIAppendixComics
% Gibt alle Sprachen aus \FDILanguageList als eigene Abschnitte mit
% Ueberschrift und Sprungmarke aus, ausser der Sprache, die bereits
% im Fliesstext steht.
\newcommand{\FDIAppendixComics}{%
  \edef\FDI@curlang{\FDILanguage}%
  \@for\FDI@lang:=\FDILanguageList\do{%
    \edef\FDI@langtmp{\FDI@lang}%
    \ifx\FDI@langtmp\FDI@curlang\else
      \expandafter\FDIComicSection\expandafter{\FDI@lang}%
    \fi
  }%
}

\makeatother

  ] at ([xshift=#1,yshift=-#2]current page.north west) {%
    {\FDILineNumberStyle\thelinenumber}%
  };%

\newcommand{\FDIHLine}[3]{%
  \draw[lightgray, line width=0.4pt]
    ([xshift=#1,yshift=-#3]current page.north west)
    --
    ([xshift=#2,yshift=-#3]current page.north west);
}

\newcommand{\FDILine}[4]{%
  \draw[black, line width=0.4pt]
    ([xshift=#1,yshift=-#2]current page.north west)
    --
    ([xshift=#3,yshift=-#4]current page.north west);
}
